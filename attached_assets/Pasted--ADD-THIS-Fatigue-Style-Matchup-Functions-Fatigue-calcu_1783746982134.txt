// Fatigue calculation
function calculateFatigue(matches) {
  const recentMatches = matches.slice(0, 7);
  const threeSetters = recentMatches.filter(m => 
    (m.score && m.score.includes("3-2")) || 
    (m.score && m.score.includes("2-1"))
  ).length;
  
  const fatigueScore = Math.max(20, 100 - (recentMatches.length * 8) - (threeSetters * 12));
  return Math.round(fatigueScore);
}
